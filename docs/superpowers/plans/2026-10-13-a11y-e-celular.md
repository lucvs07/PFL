# Acessibilidade, estados de conexão e celular mobile-first: plano de implementação

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Entregar as issues #14 (acessibilidade e estados de conexão) e #12 (polimento para celular) numa branch e num PR, com o layout do cliente reescrito mobile-first.

**Architecture:** Uma camada de plataforma (viewport, toque, hover, `dvh`) e tokens de alvo de toque/contraste ficam na base. **Tech Stack:** React 19 + Vite + Phaser, CSS puro (tokens em `styles/tokens.css`), socket.io-client, Vitest + jsdom + @testing-library/react. Sem dependências novas.

**Spec:** `docs/superpowers/specs/2026-10-13-a11y-e-celular-design.md`

`apps/client/scripts/layout-snapshot.js`:

```js
(() => {
  const path = (el) => {
    const parts = [];
    for (let n = el; n && n !== document.body; n = n.parentElement) {
      const sib = n.parentElement ? Array.from(n.parentElement.children).filter((c) => c.tagName === n.tagName) : [n];
      const cls = (typeof n.className === 'string' ? n.className : '').trim().split(/\s+/).filter(Boolean).slice(0, 2).join('.');
      parts.unshift(`${n.tagName.toLowerCase()}${cls ? '.' + cls : ''}:${sib.indexOf(n)}`);
    }
    return parts.join(' > ');
  };
  window.__pflSnapshot = () => {
    const out = {};
    for (const el of document.body.querySelectorAll('*')) {
      const r = el.getBoundingClientRect();
      if (r.width === 0 && r.height === 0) continue;
      out[path(el)] = { x: Math.round(r.x), y: Math.round(r.y + window.scrollY), w: Math.round(r.width), h: Math.round(r.height), fs: getComputedStyle(el).fontSize };
    }
    return out;
  };
})();
```

`apps/client/scripts/diff-snapshots.mjs`:

```js
import { readFileSync } from 'node:fs';

const [, , beforePath, afterPath] = process.argv;
const before = JSON.parse(readFileSync(beforePath, 'utf8'));
const after = JSON.parse(readFileSync(afterPath, 'utf8'));
const TOL = 2;
const changes = [];
for (const key of new Set([...Object.keys(before), ...Object.keys(after)])) {
  const a = before[key];
  const b = after[key];
  if (!a || !b) {
    changes.push(`${!a ? '+' : '-'} ${key}`);
    continue;
  }
  const moved = ['x', 'y', 'w', 'h'].some((k) => Math.abs(a[k] - b[k]) > TOL) || a.fs !== b.fs;
  if (moved) changes.push(`~ ${key}  ${a.x},${a.y} ${a.w}x${a.h} ${a.fs}  ->  ${b.x},${b.y} ${b.w}x${b.h} ${b.fs}`);
}
console.log(changes.length ? changes.join('\n') : 'sem diferenças');
process.exit(changes.length ? 1 : 0);
```

```ts
import { readdirSync, readFileSync } from 'node:fs';
import { join } from 'node:path';
import { fileURLToPath } from 'node:url';

const SRC = fileURLToPath(new URL('../src', import.meta.url));

export function cssFiles(dir: string = SRC): { file: string; css: string }[] {
  return readdirSync(dir, { withFileTypes: true }).flatMap((e) => {
    const p = join(dir, e.name);
    if (e.isDirectory()) return cssFiles(p);
    return e.name.endsWith('.css') ? [{ file: p.replace(SRC, 'src'), css: readFileSync(p, 'utf8') }] : [];
  });
}

export const indexHtml = (): string => readFileSync(fileURLToPath(new URL('../index.html', import.meta.url)), 'utf8');

export const stripComments = (css: string): string => css.replace(/\/\*[\s\S]*?\*\//g, '');

/** Seletores com :hover que não estão dentro de uma @media (hover: hover). */
export function ungatedHover(css: string): string[] {
  const stack: string[] = [];
  const found: string[] = [];
  let header = '';
  for (const ch of stripComments(css)) {
    if (ch === '{') {
      const h = header.trim();
      stack.push(h);
      if (!h.startsWith('@') && h.includes(':hover') && !stack.some((s) => /\(hover:\s*hover\)/.test(s))) found.push(h);
      header = '';
    } else if (ch === '}') {
      stack.pop();
      header = '';
    } else if (ch === ';') {
      header = '';
    } else {
      header += ch;
    }
  }
  return found;
}

function luminance(hex: string): number {
  const [r, g, b] = [1, 3, 5].map((i) => parseInt(hex.slice(i, i + 2), 16) / 255);
  const f = (c: number) => (c <= 0.03928 ? c / 12.92 : ((c + 0.055) / 1.055) ** 2.4);
  return 0.2126 * f(r) + 0.7152 * f(g) + 0.0722 * f(b);
}

export function contrastRatio(fg: string, bg: string): number {
  const [hi, lo] = [luminance(fg), luminance(bg)].sort((a, b) => b - a);
  return (hi + 0.05) / (lo + 0.05);
}
```

```ts
import { RECONNECT_GRACE_MS, type RoomState } from '@pfl/protocol';

/** Sem a primeira conexão por tanto tempo, o servidor provavelmente está dormindo (Render gratuito). */
export const WAKING_AFTER_MS = 4000;
/** Com o /health sem resposta, o aviso vem antes. */
export const WAKING_AFTER_HEALTH_DOWN_MS = 1500;
/** Tempo típico para o Render acordar; só orienta o texto. */
export const WAKING_ESTIMATE_MS = 60_000;

export type ConnectionState =
  | { kind: 'ok' }
  | { kind: 'waking'; elapsedMs: number }
  | { kind: 'reconnecting' }
  | { kind: 'paused' };

export interface ConnectionInput {
  connected: boolean;
  /** já houve ao menos uma conexão nesta aba */
  everConnected: boolean;
  /** quando a primeira tentativa começou (ms) */
  startedAt: number;
  now: number;
  /** o ping em /health falhou */
  healthDown: boolean;
  room: Pick<RoomState, 'status' | 'you' | 'players'> | null;
}

/** O meu lado desconectado (acordando/reconectando) vem antes da pausa por causa do adversário. */
export function connectionState(i: ConnectionInput): ConnectionState {
  if (!i.connected) {
    if (i.everConnected) return { kind: 'reconnecting' };
    const elapsedMs = Math.max(0, i.now - i.startedAt);
    const after = i.healthDown ? WAKING_AFTER_HEALTH_DOWN_MS : WAKING_AFTER_MS;
    return elapsedMs >= after ? { kind: 'waking', elapsedMs } : { kind: 'ok' };
  }
  const room = i.room;
  if (room) {
    if (room.status === 'PAUSED') return { kind: 'paused' };
    if (room.status === 'IN_MATCH' && room.players.some((p) => p.key !== room.you && !p.connected)) return { kind: 'paused' };
  }
  return { kind: 'ok' };
}

export interface ConnectionMessage {
  title: string;
  /** anunciado pelo leitor de tela; não muda a cada segundo */
  text: string;
  /** contador visual, escondido do leitor de tela */
  detail?: string;
}

export function connectionMessage(state: ConnectionState): ConnectionMessage | null {
  switch (state.kind) {
    case 'ok':
      return null;
    case 'waking':
      return {
        title: 'Servidor acordando',
        text: `O servidor gratuito dorme quando ninguém joga. Pode levar até ${WAKING_ESTIMATE_MS / 60_000} min.`,
        detail: `${Math.round(state.elapsedMs / 1000)} s`,
      };
    case 'reconnecting':
      return { title: 'Reconectando', text: 'Sem conexão com o servidor. Suas jogadas voltam quando a conexão voltar.' };
    case 'paused':
      return {
        title: 'Partida pausada',
        text: `Adversário desconectado. Aguardando a reconexão (até ${RECONNECT_GRACE_MS / 1000} s).`,
      };
  }
}
```

```ts
/**
 * Acorda o Render antes do socket. O handler de /health não envia CORS, então o pedido é `no-cors`:
 * a resposta é opaca (não dá para ler), mas resolver já prova que o servidor está de pé; rejeitar
 * quer dizer "ainda dormindo ou fora do ar". Nunca lança.
 */
export async function warmServer(url: string, fetchImpl: typeof fetch | undefined = globalThis.fetch): Promise<boolean> {
  try {
    if (!fetchImpl) return false;
    await fetchImpl(`${url.replace(/\/$/, '')}/health`, { mode: 'no-cors', cache: 'no-store' });
    return true;
  } catch {
    return false;
  }
}
```

```ts
import { useEffect, useRef, useState } from 'react';
import type { RoomState } from '@pfl/protocol';
import { useNow } from '../app/useNow';
import { connectionState, type ConnectionState } from './connectionState';
import { serverUrl, type PflSocket } from './socket';
import { warmServer } from './warmServer';

/** Liga o socket, o relógio e o ping de /health à função pura `connectionState`. */
export function useConnectionState(socket: PflSocket, room: RoomState | null): ConnectionState {
  const [connected, setConnected] = useState(socket.connected);
  const [everConnected, setEverConnected] = useState(socket.connected);
  const [healthDown, setHealthDown] = useState(false);
  const startedAt = useRef(Date.now());
  // o relógio só corre enquanto desconectado: o contador de "acordando" precisa dele
  const now = useNow(connected ? null : 1000);

  useEffect(() => {
    const onConnect = () => {
      setConnected(true);
      setEverConnected(true);
    };
    const onDisconnect = () => setConnected(false);
    socket.on('connect', onConnect);
    socket.on('disconnect', onDisconnect);
    return () => {
      socket.off('connect', onConnect);
      socket.off('disconnect', onDisconnect);
    };
  }, [socket]);

  useEffect(() => {
    let alive = true;
    void warmServer(serverUrl()).then((up) => {
      if (alive) setHealthDown(!up);
    });
    return () => {
      alive = false;
    };
  }, []);

  return connectionState({ connected, everConnected, startedAt: startedAt.current, now, healthDown, room });
}
```

```ts
/** Índice do próximo item habilitado na direção da tecla; null = fica onde está. Sem dar a volta. */
export function nextIndex(key: string, current: number, enabled: boolean[], columns: number): number | null {
  if (key === 'Home') {
    const i = enabled.indexOf(true);
    return i === -1 ? null : i;
  }
  if (key === 'End') {
    const i = enabled.lastIndexOf(true);
    return i === -1 ? null : i;
  }
  const step = STEP[key]?.(Math.max(1, columns));
  if (step === undefined) return null;
  for (let i = current + step; i >= 0 && i < enabled.length; i += step) {
    if (enabled[i]) return i;
  }
  return null;
}

/**
 * Setas, Home e End entre os botões de um grupo de escolhas. Tab continua passando por todos eles:
 * as setas são um atalho, não tabindex móvel (cada botão envia uma jogada, não é um radio).
 * `columns` omitido: conta os botões da primeira linha pela posição na tela.
 */
export function useArrowNav<T extends HTMLElement>(columns?: number): { ref: RefObject<T | null>; onKeyDown: (e: KeyboardEvent<T>) => void } {
  const ref = useRef<T | null>(null);
  const onKeyDown = useCallback(
    (e: KeyboardEvent<T>) => {
      const root = ref.current;
      if (!root) return;
      const items = Array.from(root.querySelectorAll<HTMLElement>('button'));
      const current = items.indexOf((e.target as HTMLElement).closest('button') as HTMLElement);
      if (current === -1) return;
      const cols = columns ?? Math.max(1, items.filter((el) => el.offsetTop === items[0].offsetTop).length);
      const target = nextIndex(
        e.key,
        current,
        items.map((el) => !el.hasAttribute('disabled')),
        cols,
      );
      if (target === null) return;
      e.preventDefault();
      items[target].focus();
    },
    [columns],
  );
  return { ref, onKeyDown };
}
```

```ts
/** O que um toque/clique numa carta faz. `touch` = o aparelho não tem hover; no toque o 1º toque só mostra o tooltip. */
```

```js
(() => {
  const parse = (c) => {
    const m = c.match(/rgba?\(([^)]+)\)/);
    if (!m) return null;
    const [r, g, b, a = 1] = m[1].split(',').map((x) => parseFloat(x));
    return { r, g, b, a };
  };
  const lum = ({ r, g, b }) => {
    const f = (v) => ((v /= 255) <= 0.03928 ? v / 12.92 : ((v + 0.055) / 1.055) ** 2.4);
    return 0.2126 * f(r) + 0.7152 * f(g) + 0.0722 * f(b);
  };
  const ratio = (a, b) => {
    const [hi, lo] = [lum(a), lum(b)].sort((x, y) => y - x);
    return (hi + 0.05) / (lo + 0.05);
  };
  const bgOf = (el) => {
    for (let n = el; n; n = n.parentElement) {
      const c = parse(getComputedStyle(n).backgroundColor);
      if (c && c.a > 0.9) return c;
    }
    return { r: 255, g: 255, b: 255, a: 1 };
  };
  const visible = (el) => {
    const r = el.getBoundingClientRect();
    const s = getComputedStyle(el);
    return r.width > 0 && r.height > 0 && s.visibility !== 'hidden' && s.display !== 'none';
  };

  window.__pflAudit = () => {
    const doc = document.documentElement;
    const smallTargets = [];
    for (const el of document.querySelectorAll('button, a[href], input, select, textarea, [role="button"]')) {
      if (!visible(el) || el.disabled) continue;
      const r = el.getBoundingClientRect();
      if (r.width < 44 || r.height < 44) smallTargets.push({ text: (el.textContent || el.getAttribute('aria-label') || el.tagName).trim().slice(0, 30), w: Math.round(r.width), h: Math.round(r.height) });
    }
    const lowContrast = [];
    const walker = document.createTreeWalker(document.body, NodeFilter.SHOW_TEXT);
    for (let n = walker.nextNode(); n; n = walker.nextNode()) {
      const text = n.textContent.trim();
      const el = n.parentElement;
      if (!text || !el || !visible(el) || el.closest('[aria-hidden="true"], canvas')) continue;
      const s = getComputedStyle(el);
      const fg = parse(s.color);
      if (!fg) continue;
      const bg = bgOf(el);
      const size = parseFloat(s.fontSize);
      const large = size >= 24 || (size >= 18.66 && parseInt(s.fontWeight, 10) >= 700);
      const r = ratio(fg, bg);
      if (r < (large ? 3 : 4.5)) lowContrast.push({ text: text.slice(0, 30), ratio: Math.round(r * 100) / 100, fg: s.color, bg: `rgb(${bg.r}, ${bg.g}, ${bg.b})` });
    }
    return { overflowX: doc.scrollWidth > doc.clientWidth, scrollWidth: doc.scrollWidth, clientWidth: doc.clientWidth, smallTargets, lowContrast };
  };
})();
```
